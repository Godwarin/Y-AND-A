cmake_minimum_required(VERSION 3.24)
project(yanda_android_receiver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED MEDIA_SDK)
  message(FATAL_ERROR "Pass MEDIA_SDK (existing pinned vcpkg x64-windows directory)")
endif()
add_executable(yanda-android main.cpp)
target_include_directories(yanda-android SYSTEM PRIVATE "${MEDIA_SDK}/include" "${MEDIA_SDK}/include/SDL2")
target_link_directories(yanda-android PRIVATE "${MEDIA_SDK}/lib")
target_compile_definitions(yanda-android PRIVATE SDL_MAIN_HANDLED WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(yanda-android PRIVATE /W4 /utf-8 /guard:cf)
target_link_options(yanda-android PRIVATE /guard:cf /CETCOMPAT /DYNAMICBASE /NXCOMPAT)
target_link_libraries(yanda-android PRIVATE avcodec avutil swscale SDL2 libssl libcrypto ws2_32 iphlpapi)
